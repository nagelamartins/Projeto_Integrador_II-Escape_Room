def calcular_codigo(numeros):
    codigo = 0

    for numero in numeros:
        if numero % 2 == 0:
            codigo += numero * 2
        else:
            codigo += numero

    return codigo


salas = {
    "sala1": [3, 8],
    "sala2": [5, 4],
    "sala3": [7, 6]
}

senha = 0

for sala, numeros in salas.items():
    codigo = calcular_codigo(numeros)
    senha += codigo
